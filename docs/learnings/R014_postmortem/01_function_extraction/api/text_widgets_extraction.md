```python
# =============================================================================

# Generate content for markdown, card, and form widgets
# =============================================================================

from datetime import datetime

import dspy

from api.dspy_signatures import (
    CardContentSignature,
    FormContentSignature,
    MarkdownContentSignature,
)
from api.models import UIDescriptor


class TextWidgetGenerator:
    """Generate content for text-based widgets."""

    @staticmethod
    async def generate_markdown(prompt: str) -> UIDescriptor:
        """Generate markdown content."""
        generator = dspy.Predict(MarkdownContentSignature)
        result = generator(topic=prompt)
        return UIDescriptor(
            id=f"markdown-{datetime.now().timestamp()}",
            type="markdown",
            timestamp=datetime.now().isoformat(),
            content=result.content,
            metadata={"format": "markdown"},
        )

    @staticmethod
    async def generate_card(prompt: str) -> UIDescriptor:
        """Generate card content."""
        generator = dspy.Predict(CardContentSignature)
        result = generator(topic=prompt)
        return UIDescriptor(
            id=f"card-{datetime.now().timestamp()}",
            type="card",
            timestamp=datetime.now().isoformat(),
            title=result.title,
            content=result.content,
            metadata={
                "icon": "sparkles",
                "actions": [
                    {"label": "Learn More", "action": "more", "variant": "outline"}
                ],
            },
        )

    @staticmethod
    async def generate_form(prompt: str) -> UIDescriptor:
        """Generate form content."""
        generator = dspy.Predict(FormContentSignature)
        result = generator(form_type=prompt)
        return UIDescriptor(
            id=f"form-{datetime.now().timestamp()}",
            type="form",
            timestamp=datetime.now().isoformat(),
            title=result.title,
            content=result.description,
            metadata={
                "form_id": "dynamic-form",
                "submit_label": "Submit",
                "fields": [
                    {
                        "name": "response",
                        "type": "textarea",
                        "label": "Your Response",
                        "required": True,
                        "placeholder": "Type here...",
                    }
                ],
            },
        )
```

## Detailed Analysis

### `generate_markdown(prompt)`

**Flow**:
```python
1. Create new dspy.Predict(MarkdownContentSignature)
2. Call generator(topic=prompt)
3. Return UIDescriptor with:
   - id: "markdown-{timestamp}"
   - type: "markdown"
   - content: LLM-generated markdown
   - metadata: {"format": "markdown"}
```

**What Works**:
- ✅ Simple, straightforward
- ✅ Clean output structure

**Issues**:
- ⚠️ Creates new Predict instance each call (no caching)
- ⚠️ `datetime.now().timestamp()` for ID - potential collision
- ⚠️ No error handling (LLM failures propagate)

```python
   - metadata: {"icon": "sparkles", "actions": [...]}
```

**What Works**:

**Issues**:
- ⚠️ **Hardcoded icon**: "sparkles" - not contextual
- ⚠️ **Hardcoded actions**: "Learn More" - not generated by LLM


## Behavioral Notes

### LLM Interactions

1. **Synchronous Predict**: `dspy.Predict()` is synchronous
2. **No Streaming**: Awaits complete response
3. **No Retry**: Single call only
4. **Input/Output Mapping**:
   - `prompt` → `topic` or `form_type` or `action_type`
   - Result fields → UIDescriptor fields

### Error Handling

- **None**: Errors propagate to caller
- Caller (e.g., endpoints.py) wraps in try/except
- Creates error widget on failure

### Performance

| Metric | Value | Notes |
|--------|-------|-------|
| Predict creation | ~0ms | Lightweight |
| LLM call | 500-2000ms | Depends on model/load |
| UIDescriptor creation | ~0ms | Lightweight |
| Total per call | 500-2000ms | Dominated by LLM |

## Lessons Learned

### What Works

- Consistent async interface
- Clear method naming
- Simple, predictable output

### What Doesn't Work

- **ID collision**: `timestamp()` not unique enough
- **No Predict caching**: Wasteful instantiation
- **Hardcoded metadata**: Defeats purpose of LLM
- **No error handling**: Errors propagate

### Should Copy

- Static method pattern (stateless)
- Consistent UIDescriptor structure
- Type-specific generators

### Should Avoid

- Hardcoded metadata values
- Timestamp-based IDs
- Creating Predict instances repeatedly
- Not extracting common patterns
